man evince                          # Opens the man page for the evince document viewer.
evince /home/student/passwd.ps      # Opens passwd.ps in evince using default mode.
evince -w /home/student/passwd.ps   # Opens passwd.ps in evince preview mode.
evince -i 3 /home/student/passwd.ps # Opens passwd.ps starting directly on page 3.

lp passwd.ps -P 2-3                 # Prints only pages 2 and 3 of passwd.ps to the default printer.
