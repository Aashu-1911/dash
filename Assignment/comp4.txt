lab start rhcsa-rh124-review4                                                                                                                                   # Prepares the lab environment and required resources on workstation.
sudo -i                                                                                                                                                         # Switches to root shell on serverb console (adjust interface name below if not eth0).
ip link                                                                                                                                                         # Lists network interfaces to identify the Ethernet interface name.
nmcli con show --active                                                                                                                                         # Displays active NetworkManager connection profiles and their devices.
nmcli con add con-name static-review4 ifname eth0 type ethernet ipv4.method manual ipv4.addresses 172.25.250.111/24 ipv4.gateway 172.25.250.254 ipv4.dns 172.25.250.254  # Creates static connection profile without DHCP (replace eth0 if needed).
nmcli con up static-review4                                                                                                                                     # Activates the new static-review4 connection profile.
hostnamectl set-hostname server-review4.lab4.example.com                                                                                                        # Sets the static hostname on serverb.
echo "172.25.250.10 client-review4" >> /etc/hosts                                                                                                               # Sets client-review4 as the canonical hostname for 172.25.250.10.
nmcli con mod static-review4 +ipv4.addresses 172.25.250.211/24                                                                                                    # Adds the second static IPv4 address without removing the first.
nmcli con up static-review4                                                                                                                                     # Re-activates static-review4 so serverb responds on both IP addresses.
ping -c2 172.25.250.111 && ping -c2 172.25.250.211                                                                                                                # Verifies serverb responds to both configured IPv4 addresses.
nmcli con up "System eth0"                                                                                                                                      # Restores original network settings by activating the original profile (adjust name if needed).
lab grade rhcsa-rh124-review4                                                                                                                                   # Grades the lab exercise on workstation and reports any failures.
lab finish rhcsa-rh124-review4                                                                                                                                  # Cleans up and completes the lab exercise on workstation.
